  function rng(seed) { let s = seed * 9301 + 49297; return () => { s = (s * 9301 + 49297) % 233280; return s / 233280; }; }
  function shuffle(a, r) { a = a.slice(); for (let i = a.length - 1; i > 0; i--) { const j = Math.floor(r() * (i + 1)); [a[i], a[j]] = [a[j], a[i]]; } return a; }
  const firstSentence = s => (s.match(/^.*?[.!?](?=\s|$)/) || [s])[0];
  const quizCache = {};
  function quizFor(cid) {
    if (quizCache[cid]) return quizCache[cid];
    const r = rng(cid + 7), a = ANN[cid], qs = [];
    const inCh = new Set([...Object.keys(a.updates || {}), ...Object.keys(E).filter(k => E[k].mentions[cid])]);
    const ppl = Object.keys(a.updates || {}).filter(k => k !== 'gandhi' && E[k] && E[k].kind === 'person');
    if (ppl.length) {
      const ans = ppl[Math.floor(r() * ppl.length)];
      let pool = ALL.filter(k => E[k].kind === 'person' && !inCh.has(k) && E[k].first <= cid && k !== 'gandhi');
      if (pool.length < 3) pool = ALL.filter(k => E[k].kind === 'person' && !inCh.has(k) && k !== 'gandhi');
      const opts = shuffle([ans, ...shuffle(pool, r).slice(0, 3)], r);
      qs.push({ q: 'Who plays a part in this chapter?', o: opts.map(k => short(E[k].name)), a: opts.indexOf(ans), x: E[ans].intro });
    }
    const pl = (a.where || []).filter(k => E[k]);
    if (pl.length) {
      const ans = pl[0];
      let pool = ALL.filter(k => E[k].kind === 'place' && !(a.where || []).includes(k) && E[k].first <= cid);
      if (pool.length < 3) pool = ALL.filter(k => E[k].kind === 'place' && !(a.where || []).includes(k));
      const opts = shuffle([ans, ...shuffle(pool, r).slice(0, 3)], r);
      qs.push({ q: 'Where does this chapter mostly take place?', o: opts.map(k => short(E[k].name)), a: opts.indexOf(ans), x: E[ans].intro });
    }
    const others = shuffle([...Array(N).keys()].filter(i => Math.abs(i - cid) >= 3 && ANN[i].summary), r).slice(0, 3);
    const opts = shuffle([cid, ...others], r);
    qs.push({ q: 'Which of these happens in this chapter?', o: opts.map(i => firstSentence(ANN[i].summary)), a: opts.indexOf(cid), x: 'That moment comes from Ch. ' + num(cid) + '.' });
    return (quizCache[cid] = qs);
  }
  function quizHTML(cid) {
    const qs = quizFor(cid); if (!qs.length) return '';
    const st = S.quiz[cid];
    let h = `<section class="quiz" data-quiz="${cid}"><h2>Chapter quiz · ${qs.length} questions · +20 XP each</h2>`;

        h += `<button class="opt ${cls}" data-o="${oi}" ${ans != null ? 'disabled' : ''}>${esc(o)}</button>`;
      });
      h += `</div>`;
    });

  function wireQuiz(root, cid) {
    root.querySelectorAll(`[data-quiz="${cid}"] .opt`).forEach(b => b.onclick = () => {
